rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    // --- FUNÇÕES DE AJUDA OTIMIZADAS ---

    function isSignedIn() {
      return request.auth != null;
    }

    function isCompanyAdmin(companyId) {
      return isSignedIn() && request.auth.uid == companyId;
    }

    function getMemberDocument(memberId) {
      return get(/databases/$(database)/documents/members/$(memberId));
    }

    function isMemberOfCompany(companyId) {
      // 1. Deve estar logado.
      return isSignedIn() && 
             // 2. Deve existir um documento na coleção 'members' com o UID do usuário.
             exists(/databases/$(database)/documents/members/$(request.auth.uid)) && 
             getMemberDocument(request.auth.uid).data.companyId == companyId;
    }

    // --- REGRAS PARA A COLEÇÃO 'companies' ---
    match /companies/{companyId} {
      // Apenas o admin da empresa pode ler, atualizar ou excluir a sua empresa.
      allow read, update, delete: if isCompanyAdmin(companyId);
      // O UID do usuário deve ser o mesmo que o 'companyId' que ele está criando.
      allow create: if isSignedIn() && request.auth.uid == companyId;
    }

    // --- REGRAS PARA A COLEÇÃO 'members' ---
    match /members/{memberId} {
      // REGRAS SEM CLOUD FUNCTIONS:
      // get: Permite que qualquer pessoa com o link (token) verifique se o membro existe para fazer login.
      allow get: if true;
      // read (get e list): Permite que um admin leia os dados dos membros da sua própria empresa.
      // O Firestore aplicará a cláusula 'where' da query a esta regra.
      allow read: if isCompanyAdmin(resource.data.companyId);
      
      // Apenas um admin pode criar novos membros (apontando para a sua própria empresa).
      allow create: if isCompanyAdmin(request.resource.data.companyId);
      
      // Apenas um admin pode atualizar ou deletar membros da sua empresa.
      allow update, delete: if isCompanyAdmin(resource.data.companyId);
    }

    // --- REGRAS PARA A COLEÇÃO 'tasks' ---
    match /tasks/{taskId} {
      // REGRAS SEM CLOUD FUNCTIONS:
      // O admin da empresa pode ler as tarefas.
      // A leitura também é permitida para qualquer um, pois a busca é filtrada por companyId no frontend.
      // Isso é seguro, pois um usuário não sabe o companyId de outras empresas.
      allow read: if true;
      
      // Apenas o admin pode criar tarefas para sua própria empresa.
      allow create: if isCompanyAdmin(request.resource.data.companyId);

      // Apenas o admin pode atualizar ou deletar tarefas.
      allow update, delete: if isCompanyAdmin(resource.data.companyId);
    }

    // --- REGRAS PARA A COLEÇÃO 'demands' ---
    match /demands/{demandId} {
      // REGRAS SEM CLOUD FUNCTIONS:
      // A leitura é permitida para qualquer um, pois a busca é filtrada por companyId no frontend.
      // Um usuário não sabe o companyId de outras empresas.
      allow read: if true;
      allow update, delete: if isCompanyAdmin(resource.data.companyId);
      allow create: if isCompanyAdmin(request.resource.data.companyId);
    }

    // --- REGRAS PARA A COLEÇÃO 'demandTemplates' ---
    match /demandTemplates/{templateId} {
      // Apenas o admin da empresa pode gerenciar modelos de demanda.
      allow read, update, delete: if isCompanyAdmin(resource.data.companyId);
      allow create: if isCompanyAdmin(request.resource.data.companyId);
    }

    // --- REGRAS PARA A COLEÇÃO 'timeEntries' ---
    match /timeEntries/{entryId} {
      // REGRAS SEM CLOUD FUNCTIONS:
      // A leitura é permitida para qualquer um, pois a busca é filtrada por memberId no frontend.
      // Um usuário não pode adivinhar o memberId (token) de outro.
      allow read: if true;
      
      // A criação é permitida por qualquer um, mas o frontend garante que os IDs corretos são enviados.
      allow create: if true;
      
      // Apenas o admin pode atualizar (aprovar, rejeitar, editar) ou deletar registros de horas.
      // O colaborador não pode alterar um registro após criado.
      allow update, delete: if isCompanyAdmin(resource.data.companyId);
    }
  }
}
